#02
#include <stdio.h>

int maiorValor(int *a, int *b, int *c){
    int maior = *a;
    if (*b > maior){
        maior = *b;
    }
    if (*c > maior){
        maior = *c;
    }
    
    return maior;
}

int main(void){
    int x, y, z;
    
    printf("Digite os valores inteiros de a, b e c: ");
    scanf("%d %d %d", &x, &y, &z);
    
    int maior = maiorValor(&x, &y, &z);
    
    printf("Maior valor: %d", maior);
    
    return 0;
}




#03

#include <stdio.h>

void result(int *a, int *b, int *soma){
    for (int i = 0; i < 5; i++){
        *(soma + i) = *(a + i) + *(b + i);
    }
}

int main(void){
    int y[4], z[4], resultado[4];
    
    printf("Digite 5 valores para o primeiro vetor:\n");
    for (int i = 0; i < 5; i++) {
        scanf("%d", &y[i]);
    }
    
    printf("Digite 5 valores para o segundo vetor:\n");
    for (int i = 0; i < 5; i++) {
        scanf("%d", &z[i]);
    }
    
    result(y, z, resultado);
    
    printf("Resultado:\n");
    for (int i = 0; i < 5; i++) {
        printf("%d", resultado[i]);
        printf("\n");
    }
    
    printf("\n");
    
    return 0;
}
