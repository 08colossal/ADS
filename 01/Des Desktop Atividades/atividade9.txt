#include <stdio.h>

int main() {
    int matriz[5][3]; //dias e horarios 
    int i,j;
    float media[5];
    float med[3];
    float maior=0, menor=0;
    int melhor=0, pior=0;
    int m5=0;
    
    for (i=0; i<5; i++){
        printf("Cliente %d.\n",i+1);
        for (j=0; j < 3; j++){
            printf("Critério %d: ", j+1);
            scanf("%d", &matriz[i][j]);
            media[i] = media[i] + matriz[i][j];
            
            if (matriz[i][j] > maior){
                maior = matriz[i][j];
                melhor = j;
            }
            if (matriz[i][j] < menor){
                menor = matriz[i][j];
                pior = j;
            }
            if (matriz[i][j] < 5){
                m5++;
            }
        }
        printf("\n\n");
    }
    
    
    //media clientes
    for (i=0; i<5; i++){
        for(j=0; j<3; j++){
            printf("%d ", matriz[i][j]);
        }
        printf("\n");
    }
    for (i = 0; i < 5; i++){
        printf("\nA média de satisfação do cliente %d é: %.1f!\n", i+1, media[i]/3);
    }
    
    //media criterios 
    for (i = 0; i < 3; i++){
        for (j=0; j<5; j++){
            med[i] = med[i] + matriz[j][i];
        }
    }
    printf("\n");
    for (i = 0; i < 3; i++){
        if (med[i]/5 >= 7){
            printf("\nA média de satisfação no critério %d é: %.1f. O cliente ficou satisfeito!\n", i+1, med[i]/5);
        }
        else{
            printf("\nA média de satisfação no critério %d é: %.1f. O cliente ficou insatisfeito!\n", i+1, med[i]/5);
        }
    }
    
    printf("\nO critério melhor avaliado foi: %d, com nota %.0f, o pior foi %d, com nota %.0f e %d notas foram menor que 5.", melhor+1, maior, pior+1, menor, m5);
    
    return 0;
}

#include <stdio.h>

int main() {
    int matriz[5][5] = {0}; //reserva de poltronas 
    int opc = 1, reserva, fila;
    
    while(opc != 0){
        
        printf("Menu\n1 - Reserva poltrona\n2 - Cancelar reserva\n3 - Exibir mapa\n4 - Mostrar relatório\n0 - sair ");
        scanf("%d", &opc);
        
        switch(opc){
            
            case 1:
                printf("Que fileira quer reservar?");
                scanf("%d", &fila);
                printf("Que poltrona quer reservar?");
                scanf("%d", &reserva);
                
                if (fila < 1 || fila > 5 || reserva < 1 || reserva > 5) {
                    printf("Posicao invalida! Escolha valores entre 1 e 5.\n");
                }
                
                else if (matriz[fila-1][reserva-1] == 0){
                    matriz[fila-1][reserva-1] = 1;
                }
                
                else{
                    printf("Já está reservada!");
                }
                
                break;
            
            
            case 2:
                printf("Que fileira quer cancelar?");
                scanf("%d", &fila);
                printf("Que poltrona quer cancelar?");
                scanf("%d", &reserva);
                
                if (fila < 1 || fila > 5 || reserva < 1 || reserva > 5) {
                    printf("Posicao invalida! Escolha valores entre 1 e 5.\n");
                }
                
                else if (matriz[fila][reserva] == 1){
                    matriz[fila-1][reserva-1] = 0;
                }
                
                else{
                    printf("Não está reservada! ");
                }
                
                break;
            
            
            case 3:
                for (int i=0; i<5; i++){
                    for(int j=0; j<5; j++){
                        printf("%d ", matriz[i][j]);
                    }
                     printf("\n");
                }

                break;
            
            
            case 4:
                int full =0, null = 0, poggers = 0;
                int reservas_max = 0;
                
                for (int i = 0; i < 5; i++){
                    int reservas_fileira = 0;
                    for (int j = 0; j < 5; j++){
                        if (matriz[i][j] == 0){
                            null++;
                        }
                        
                        else if (matriz[i][j] == 1){
                            full++;
                            reservas_fileira++;
                        }
                        
                        if (reservas_fileira > reservas_max) {
                        reservas_max = reservas_fileira;
                        poggers = i;
                    }
                    }
                }
                printf("Estão ocupadas %d cadeiras, e %d desocupadas", full, null);
                printf("\nA fileira %d tem mais reservas (total:%d).", poggers+1, reservas_max);
                break;
            
            case 0:
                printf("Encerrado.");
                
                
            default:
                printf("Opção inválida.");
                break;
        }
    }
}
