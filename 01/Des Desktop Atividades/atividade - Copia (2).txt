#include <stdio.h>

int main() {
 //idade, altura, nome
    int idade;
    float altura;
    char nome[50];
    char simounao;
    
    printf("Qual sua idade? Sua altura? Seu nome:\n");
    scanf("%d %f %s", &idade, &altura, nome);
    
    printf("Olá %s, você tem %.1f metros de altura e %d anos, correto?\nS para sim e N para não\n", nome, altura, idade);
    scanf(" %c", &simounao);
    
    if (simounao == 'S' || simounao == 's') {
        printf("Tudo certo então!");
    }
    else if (simounao == 'N' || simounao == 'n'){
        printf("Responda novamente");
    }
    else{
        printf("Resposta inválida");
    }
    
    return 0;
}
