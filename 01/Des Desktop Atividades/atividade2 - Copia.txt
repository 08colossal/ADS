//mercado: pedir nome produto e preço, classificar 
int main()
{
    char nome[50];
    float valor;
    
    printf("Qual o nome do produto?");
    scanf("%s", nome);
    printf("Qual o valor do produto?");
    scanf("%f", &valor);
    
    if (valor < 20){
        printf("o produto %s é barato", nome);
    }
    else if (valor < 100){
        printf("O produto %s é mediano", nome);
    }
    else{
        printf("Valor extravagante");
    }

    return 0;
}
