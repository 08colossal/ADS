#include <stdio.h>
#define Length 5

//5 notas, media e reprovado aprovado
int main()
{
   //vetores
   int vetor[Length];
   int soma = 0;
   
   for (int i = 0; i < Length; i++){
       printf("Qual sua nota?");
       scanf("%d", &vetor[i]);
       soma += vetor[i];
   }
   if (soma/Length >= 7){
    printf("A sua média é %d, você está aprovado.", soma/Length);
   }
   else if(5<soma/Length>6,9){
    printf("A sua média é %d, você está em recuperação.", soma/Length);
   }
   else{
    printf("A sua média é %d, você está reprovado.", soma/Length);

   }
   
    return 0;
}

#include <stdio.h>
#define Length 6

//6 temperaturas, print maior menor todas e quantas menores q 30
int main()
{
   //vetores
   float vetor[Length];
   float maior = 0, menor = 0;
   int mt = 0;
   
   for (int i = 0; i < Length; i++){
       printf("Qual a temperatura?");
       scanf("%f", &vetor[i]);
       if (i==0){
           maior = menor = vetor[i];
       }
       if(vetor[i]>maior){
           maior = vetor[i];
       }
       if(vetor[i]<menor){
           menor = vetor[i];
       }
       if(vetor[i] > 30){
            mt++;
       }
   }
   printf("A maior temperatura é %.0f, a menor é %.0f, têm %.0f menores que 30º. Essas são todas as temperaturas: ", maior, menor, mt);
   for (int i = 0; i < Length; i++){
       printf("%.0f ", vetor[i]);
   }
    return 0;
}

#include <stdio.h>
#define Length 8

//8 estoques de produtos
int main()
{
   //vetores
   int vetor[Length];
   int maior = 0, menor = 0;
   int med = 0, baixo = 0, normal = 0, alto = 0, acima = 0;
   
   for (int i = 0; i < Length; i++){
       printf("Quantas unidades tem no estoque do produto %d?", i+1);
       scanf("%d", &vetor[i]);
       if ( i == 0){
           maior = menor = vetor[i];
       }
       if (vetor[i] > maior){
           maior = vetor[i];
       }
       if(vetor[i]< menor){
           menor = vetor[i];
       }
       
       if (vetor[i] < 5){
            baixo++;
         }
        else if (5 < vetor[i] > 20){
            normal++;
        }
        else{
            alto++;
        }
        med = med + vetor[i];
    }
    med = med / Length;
   
   for ( int i = 0; i < Length; i++){
       if (vetor[i] > med){
           acima++;
       }
   }
   printf("Tem %d produtos no estoque. A média de unidades é %d, a maior quantia armazenada é %d e a menor é %d. Tem %d produtos com estoque baixo, %d com estoque normal e %d com estoque acima da média.\n", Length, med, maior, menor, baixo, normal, alto, acima);
   
   for (int i = 0; i < Length; i++){
       if (vetor[i] < 5){
            printf("Produto %d: %d unidades - estoque baixo\n", i+1, vetor[i]);
        }
        else if (5 < vetor[i] > 20){
             printf("Produto %d: %d unidades - estoque normal\n", i+1, vetor[i]);
         }
        else{
             printf("Produto %d: %d unidades - estoque alto\n", i+1, vetor[i]);
         }
      
   }
    return 0;
}
